**Step 2: Cross-pack integrity verification.**

Verify that every pack's `tasks[].follow_up_tree` values have matching `follow_up_trees[].task_type` entries (cross-reference integrity):

```bash
python -c "
import yaml, sys
from pathlib import Path

source_dir = Path('prompt_tool/knowledge_packs')
packs = ['02-sales-retail.yaml', '03-education.yaml', '04-finance.yaml', '05-manufacturing.yaml']
all_ok = True

for pack_file in packs:
    path = source_dir / pack_file
    data = yaml.safe_load(open(path, encoding='utf-8'))
    pack_name = data['meta']['name']
    
    # Collect tree task_types
    tree_types = {t['task_type'] for t in data.get('follow_up_trees', [])}
    
    # Check every task's follow_up_tree references a real tree
    for task in data.get('tasks', []):
        ref = task.get('follow_up_tree', '')
        if ref and ref not in tree_types:
            print(f'{pack_name}: task \"{task[\"name\"]}\" references missing tree \"{ref}\"')
            all_ok = False
    
    # Check for duplicate node_ids across trees in the same pack
    all_node_ids = []
    for tree in data.get('follow_up_trees', []):
        for nid in tree.get('nodes', {}):
            all_node_ids.append(nid)
    dupes = [n for n in set(all_node_ids) if all_node_ids.count(n) > 1]
    if dupes:
        print(f'{pack_name}: DUPLICATE node_ids across trees: {dupes}')
        all_ok = False
    
    print(f'{pack_name}: {len(data[\"tasks\"])} tasks × {len(tree_types)} trees — cross-ref OK' if not any(t.get('follow_up_tree','') and t['follow_up_tree'] not in tree_types for t in data.get('tasks',[])) else '')

if not all_ok:
    print()
    print('CROSS-REFERENCE INTEGRITY CHECK FAILED')
    sys.exit(1)
print()
print('Cross-reference integrity PASSED')
"
```
